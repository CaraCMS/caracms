import { bucket, defineRailway, github, image, preserve, project, redis, ref, service } from "railway/iac";

// A caracms site's Railway project, as code (written by `caracms init` from the
// caracms starter's). From the site's root: `railway config plan` shows
// what would change, `railway config apply` makes it so. Plan first, every time:
// Redis holds the editor's drafts and the media bucket every uploaded image.
//
// What this file doesn't do, set by hand on Railway: create the project, the
// domains (site's and imgproxy's, imgproxy's on port 8080),
// Redis's volume and start command (Railway's Redis template makes them), the
// secrets' values, and the optional secrets below until they're wanted.

// ---- The site's names: a new site edits this block and nothing else. ----
// Project name, bucket names and the S3 bucket references derive from SITE.
const SITE = "{{SITE}}";
// The repo the editor commits to and Railway builds from.
const GITHUB_REPO = "{{GITHUB_REPO}}";
// Railway's region for the services, and the buckets' region (a bucket region
// is one of sjc, iad, ams, sin; ams sits next to europe-west4).
const REGION = "europe-west4-drams3a";
const BUCKET_REGION = "ams";
// The site is the repo's root: no watch paths, so every push to main deploys.
const WATCH_PATHS: string[] = [];
const BUILD_COMMAND = "{{BUILD}}";
const START_COMMAND = "{{START}}";
// imgproxy's origin and the media bucket's S3 name. Null: they come as
// references to imgproxy's domain and the media bucket's BUCKET.
const MEDIA_URL: string | null = null;
const MEDIA_S3_NAME: string | null = null;
// -------------------------------------------------------------------------

// A Railway reference inside a longer string: ${{<resource>.<VARIABLE>}}.
// ref() is for a value that is one whole reference; this is for one wrapped in
// other text. ref() compiles to the same syntax from the resource's name.
const inline = (resource: { name: string }, variable: string) =>
  "${{" + resource.name + "." + variable + "}}";

export default defineRailway(() => {
  // Uploaded images, served publicly through imgproxy.
  const media = bucket(`${SITE}-media`, { region: BUCKET_REGION });
  // Form submissions: a second bucket, so imgproxy, which holds the media
  // bucket's keys and serves it publicly, can never reach them. Only `site`
  // gets its credentials (SUBMISSIONS_S3_*). Same region as media: the site
  // uses media's endpoint and region (S3_ENDPOINT, S3_REGION) for both.
  const submissions = bucket(`${SITE}-submissions`, { region: BUCKET_REGION });

  // The editor's drafts and sessions.
  const kv = redis("Redis", { region: REGION });
  // Sleeps when idle. redis() takes only a region, so it's set on the node.
  kv.deploy = { ...kv.deploy, sleepApplication: true };

  const imgproxy = service("imgproxy", {
    // Pinned to an exact release, no auto updates.
    source: image("ghcr.io/imgproxy/imgproxy:v4.0.15"),
    replicas: { [REGION]: 1 },
    // Sleeps when idle.
    deploy: { sleepApplication: true },
    // Railway's edge caches the signed image URLs (immutable, a year).
    cdn: { enabled: true },
    env: {
      // The domain is generated for this port, so it's fixed here.
      PORT: "8080",
      IMGPROXY_USE_S3: "true",
      IMGPROXY_S3_ENDPOINT: ref(media, "ENDPOINT"),
      IMGPROXY_S3_REGION: ref(media, "REGION"),
      AWS_ACCESS_KEY_ID: ref(media, "ACCESS_KEY_ID"),
      AWS_SECRET_ACCESS_KEY: ref(media, "SECRET_ACCESS_KEY"),
      // Only the media bucket: BUCKET is the S3 name with Railway's suffix.
      IMGPROXY_ALLOWED_SOURCES: `s3://${MEDIA_S3_NAME ?? inline(media, "BUCKET")}/`,
      IMGPROXY_MAX_RESULT_DIMENSION: "4000",
      IMGPROXY_MAX_SRC_RESOLUTION: "50",
      IMGPROXY_TTL: "31536000",
      IMGPROXY_USE_ETAG: "true",
      IMGPROXY_WORKERS: "2",
      IMGPROXY_MALLOC: "jemalloc",
      MALLOC_ARENA_MAX: "2",
      IMGPROXY_SKIP_PROCESSING_FORMATS: "gif,svg",
      // Signing key and salt (64-byte hex, generated by hand). `site` refers
      // to these, so the two can't drift apart.
      IMGPROXY_KEY: preserve(),
      IMGPROXY_SALT: preserve(),
    },
  });

  const site = service("site", {
    source: github(GITHUB_REPO, { branch: "main", checkSuites: false }),
    build: { buildCommand: BUILD_COMMAND, watchPatterns: WATCH_PATHS },
    start: START_COMMAND,
    healthcheck: "/health",
    replicas: { [REGION]: 1 },
    deploy: { restartPolicyMaxRetries: 3 },
    env: {
      // ---- Written here: literals and Railway references ----
      NODE_ENV: "production",
      // The public address (canonical links, the sitemap, og:url). At launch it
      // becomes the client's domain, written out as "https://<domain>".
      SITE_URL: "https://${{RAILWAY_PUBLIC_DOMAIN}}",
      REDIS_URL: ref(kv, "REDIS_URL"),
      // Images: imgproxy's public origin (the custom image host at launch), and
      // its key and salt, which sign every image URL.
      CARACMS_MEDIA_URL: MEDIA_URL ?? `https://${inline(imgproxy, "RAILWAY_PUBLIC_DOMAIN")}`,
      IMGPROXY_KEY: ref(imgproxy, "IMGPROXY_KEY"),
      IMGPROXY_SALT: ref(imgproxy, "IMGPROXY_SALT"),
      // The media bucket, for uploads from the editor.
      S3_ENDPOINT: ref(media, "ENDPOINT"),
      S3_REGION: ref(media, "REGION"),
      S3_BUCKET: ref(media, "BUCKET"),
      S3_ACCESS_KEY_ID: ref(media, "ACCESS_KEY_ID"),
      S3_SECRET_ACCESS_KEY: ref(media, "SECRET_ACCESS_KEY"),
      // The submissions bucket. Its endpoint and region are media's (S3_*).
      SUBMISSIONS_S3_BUCKET: ref(submissions, "BUCKET"),
      SUBMISSIONS_S3_ACCESS_KEY_ID: ref(submissions, "ACCESS_KEY_ID"),
      SUBMISSIONS_S3_SECRET_ACCESS_KEY: ref(submissions, "SECRET_ACCESS_KEY"),

      // The repo the editor commits to.
      GITHUB_REPO,

      // ---- Set by hand on Railway: names only, the values never live here ----
      // On a fresh project these stay unset and the plan doesn't mention them.
      // The editor's GitHub App and sign-in.
      GITHUB_APP_ID: preserve(),
      GITHUB_APP_PRIVATE_KEY: preserve(),
      GITHUB_WEBHOOK_SECRET: preserve(),
      CARACMS_SECRET: preserve(),
      CARACMS_ALLOW: preserve(),
      CARACMS_BACKUP_USER: preserve(),
      CARACMS_BACKUP_PASSWORD: preserve(),
      // Try again on a failed publish redeploys through this project token.
      CARACMS_RAILWAY_TOKEN: preserve(),
      MUX_TOKEN_ID: preserve(),
      MUX_TOKEN_SECRET: preserve(),
      // The form emails.
      RESEND_API_KEY: preserve(),
      FORMS_FROM: preserve(),
      // Optional. The form's spam check: the site key is baked in at build, so
      // a change needs a redeploy.
      TURNSTILE_SITE_KEY: preserve(),
      TURNSTILE_SECRET_KEY: preserve(),
      // Optional. Pre-launch password ("user:password"); unset means public.
      CARACMS_PRELAUNCH: preserve(),
    },
  });

  return project(SITE, {
    resources: [site, kv, imgproxy, media, submissions],
  });
});
